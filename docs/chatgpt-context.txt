namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\QuoteRequest;
use Filament\Facades\Filament;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;

// Download a customer's uploaded list (private disk) – admins only

{
    public function __invoke(Request $request, QuoteRequest $quote)
    {
        abort_unless($request->user()?->canAccessPanel(Filament::getPanel('admin')) && $request->user()->can('view', $quote), 403);

            $quote->attachment_path,
            $quote->attachment_name ?: basename($quote->attachment_path),
            ['X-Content-Type-Options' => 'nosniff', 'Content-Security-Policy' => "default-src 'none'"]
        );
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Product;
use App\Support\Cart;
use Illuminate\Http\Request;

class CartController extends Controller
{
    public function index()
    {
        $products = Cart::items();
        $total    = $products->sum('cart_subtotal');

        return view('cart.index', compact('products', 'total'));
    }

    public function add(Request $request, Product $product)
    {
        // Quantity from request (fallback to 1, never < 1)
        $quantity = max(1, (int) $request->input('quantity', 1));

        // If this product is already in the cart, increase quantity
        Cart::put($product, Cart::quantityOf($product) + $quantity);

        $cartCount = Cart::count();

        // If this is an AJAX/JSON request (our fetch call), return JSON
        if ($request->expectsJson()) {
            return response()->json([
                'success'    => true,
                'message'    => 'Product added to cart',
                'cart_count' => $cartCount,
                'product'    => ['id' => $product->id, 'name' => $product->name, 'price' => $product->getDisplayPrice(), 'quantity' => $quantity],
            ]);
        }

        // Fallback for normal form post
        return redirect()->back()->with('success', 'Product added to cart');
    }

    public function update(Request $request, Product $product)
    {
        $qty = max(0, (int) $request->input('quantity', 1));

        // 0 removes the line
        Cart::put($product, $qty);

        return back()->with('success', 'Cart updated.');
    }

    public function remove(Product $product)
    {
        Cart::remove($product);

        return back()->with('success', 'Product removed from cart.');
    }

    public function clear()
    {
        Cart::clear();

        return back()->with('success', 'Cart cleared.');
    }
}

<?php

namespace App\Http\Controllers;

use App\Actions\PlaceOrderRequest;
use App\Http\Requests\CheckoutRequest;
use App\Support\Cart;
use App\Support\OrderHistory;
use Illuminate\Support\Facades\RateLimiter;

class CheckoutController extends Controller
{
    public function show()
    {
        $products = Cart::items();

        if ($products->isEmpty()) {
            return redirect()->route('cart.index')->with('error', 'Your cart is empty. Add some products before checking out.');
        }

        return view('checkout.show', [
            'products' => $products,
            'total'    => $products->sum('cart_subtotal'),
        ]);
    }

    public function store(CheckoutRequest $request, PlaceOrderRequest $placeOrder)
    {
        // Honeypot: bots fill the hidden field – pretend it worked, create nothing
        if ($request->filled('website')) {
            Cart::clear();

            return redirect()->route('home')->with('success', 'Thank you, your order request has been received.');
        }

        $customer = $request->customer();

        // Abuse limits (counted on successful requests only)
        $ipKey    = 'order-request:ip:' . $request->ip();
        $phoneKey = 'order-request:phone:' . $customer['customer_phone'];

        if (RateLimiter::tooManyAttempts($ipKey, config('orders.limits.per_ip_per_hour'))
            || RateLimiter::tooManyAttempts($phoneKey, config('orders.limits.per_phone_per_hour'))) {
            return back()
                ->withErrors(['form' => 'You have sent several order requests recently. Please wait a while, or call/WhatsApp us directly.'])
                ->withInput();
        }

        $products = Cart::items();

        if ($products->isEmpty()) {
            return redirect()->route('cart.index')->with('error', 'Your cart is empty. Add some products before checking out.');
        }

        $order = $placeOrder->handle($customer, $products, $request->ip(), $request->userAgent());

        RateLimiter::hit($ipKey, 3600);
        RateLimiter::hit($phoneKey, 3600);

        Cart::clear();
        OrderHistory::remember($request, $order);

        return redirect()
            ->route('orders.show', $order)
            ->with('order_placed', true);
    }
}

<?php

namespace App\Http\Controllers;

use App\Support\ErrorRecorder;
use Illuminate\Http\Request;

// Receives error reports from visitors' browsers (resources/js/app.js) → Error log
class ClientErrorController extends Controller
{
    public function store(Request $request)
    {
        $data = $request->validate([
            'type'    => ['nullable', 'string', 'max:100'],
            'level'   => ['nullable', 'in:error,warning'],
            'message' => ['required', 'string', 'max:2000'],
            'source'  => ['nullable', 'string', 'max:500'],
            'line'    => ['nullable', 'integer', 'min:0'],
            'stack'   => ['nullable', 'string', 'max:8000'],
            'page'    => ['nullable', 'string', 'max:2000'],
            'details' => ['nullable', 'array', 'max:20'],
        ]);

        // Browser noise we can't act on (extensions, cross-origin scripts)
        if (preg_match('/^(Script error\.?|ResizeObserver loop)/i', $data['message'])) {
            return response()->noContent();
        }

        ErrorRecorder::browser($data);

        return response()->noContent();
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\ContactMessage;
use App\Models\SiteSetting;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Mail;
use Illuminate\Support\Facades\RateLimiter;

class ContactController extends Controller
{
    public function index()
    {
        $settings = SiteSetting::current();

        return view('contact', compact('settings'));
    }

    public function submit(Request $request)
    {
        // Honeypot field to catch bots
        if ($request->filled('website')) {
            // silently ignore or pretend success
            return back()->with('success', 'Thank you, your message has been received.');
        }

        // Rate limit: max 5 attempts per 10 minutes per IP
        $key = 'contact:' . $request->ip();
        if (RateLimiter::tooManyAttempts($key, 5)) {
            $seconds = RateLimiter::availableIn($key);
            return back()
                ->withErrors(['form' => "Too many attempts. Please try again in {$seconds} seconds."])
                ->withInput();
        }
        RateLimiter::hit($key, 600); // 10 minutes

        // Validate inputs (basic restrictions)
        $validated = $request->validate([
            'name'    => ['required', 'string', 'max:255'],
            'email'   => ['required', 'email', 'max:255'],
            'phone'   => ['nullable', 'string', 'max:50'],
            'subject' => ['nullable', 'string', 'max:255'],
            'message' => ['required', 'string', 'min:10', 'max:5000'],
        ]);

        // Extra sanitization: strip HTML tags to avoid injection/XSS in stored data & emails
        $sanitized = array_map(function ($value) {
            return is_string($value) ? strip_tags($value) : $value;
        }, $validated);

        $settings = SiteSetting::current();
        $recipient = $settings->contact_recipient_email ?: config('orders.admin_email');

        // Save to DB (using sanitized values)
        $contactMessage = ContactMessage::create([
            'name'       => $sanitized['name'],
            'email'      => $sanitized['email'],
            'phone'      => $sanitized['phone'] ?? null,
            'subject'    => $sanitized['subject'] ?? null,
            'message'    => $sanitized['message'],
            'ip_address' => $request->ip(),
            'user_agent' => substr((string) $request->userAgent(), 0, 1000),
        ]);

        // Send email (safe, escaped output in the mailable view)
        // The message is saved either way (Admin → Contact messages); email is a bonus.
        // Failures are reported (Error log) but never shown to the visitor.
        if ($recipient) {
            try {
                Mail::to($recipient)->send(new \App\Mail\ContactFormSubmitted($contactMessage));
            } catch (\Throwable $e) {
                report($e);
            }
        }

        return back()->with('success', 'Thank you for contacting us. We will get back to you soon.');
    }
}

<?php

namespace App\Http\Controllers;

use App\Support\ErrorRecorder;
use Illuminate\Http\Request;
use Illuminate\Support\Str;

// Receives Content-Security-Policy violation reports from browsers → Error log (warning)
class CspReportController extends Controller
{
    public function store(Request $request)
    {
        $payload = json_decode((string) $request->getContent(), true);

        // Old format: {"csp-report": {...}} – Reporting API: [{"type": "csp-violation", "body": {...}}]
        $reports = isset($payload['csp-report'])
            ? [$payload['csp-report']]
            : collect(is_array($payload) ? $payload : [])->where('type', 'csp-violation')->pluck('body')->all();

        foreach (array_slice($reports, 0, 5) as $report) {
            if (! is_array($report)) {
                continue;
            }

            $blocked   = (string) ($report['blocked-uri'] ?? $report['blockedURL'] ?? '');
            $directive = (string) ($report['effective-directive'] ?? $report['violated-directive'] ?? $report['effectiveDirective'] ?? '');

            // Browser extensions inject their own code – not something the site can fix
            if (preg_match('/^(chrome|moz|safari|ms-browser)-extension|^about:/i', $blocked)) {
                continue;
            }

            ErrorRecorder::browser([
                'type'    => 'CSP violation',
                'level'   => 'warning',
                'message' => Str::limit("Blocked {$blocked} ({$directive})", 500),
                'page'    => Str::limit((string) ($report['document-uri'] ?? $report['documentURL'] ?? ''), 2000, ''),
                'source'  => (string) ($report['source-file'] ?? $report['sourceFile'] ?? ''),
                'line'    => $report['line-number'] ?? $report['lineNumber'] ?? null,
            ]);
        }

        return response()->noContent();
    }
}

    {

            ->orderBy('sort_order')
            ->limit(3)
            ->get();

        $limit  = max(1, min(12, (int) content('home.block_limit', 8)));
        $active = fn () => Product::with('company')->where('is_active', true);

        // Product blocks (Website → Home page): each can be switched off
        $deals = content('home.deals_enabled')
            ? $active()->where('is_on_sale', true)->where('sale_price', '>', 0)->latest('updated_at')->take($limit)->get()

            : collect();

        $brands = content('home.brands_enabled')
            ? Company::where('is_active', true)
                ->whereHas('products', fn ($q) => $q->where('is_active', true))
                ->orderBy('name')
                ->get()
            : collect();

        // Category rows (Website → Home sections)

            ->where('is_active', true)

                ->where('is_active', true)
                ->where('category_id', $section->category_id)
                ->latest()
                ->take(8)
                ->get();

<?php

namespace App\Http\Controllers;

use App\Models\Order;
use App\Models\Product;
use App\Support\Cart;
use App\Support\OrderHistory;
use Illuminate\Http\Request;

// Customer-facing order pages. Access is by the secret token (no login); pages are never indexed.
class OrderController extends Controller
{
    public function index(Request $request)
    {
        $tokens = OrderHistory::tokens($request);

        $orders = empty($tokens)
            ? collect()
            : Order::withCount('items')
                ->whereIn('public_token', $tokens)
                ->latest()
                ->get();

        return response()
            ->view('orders.index', compact('orders'))
            ->withHeaders($this->privateHeaders());
    }

    public function show(Request $request, Order $order)
    {
        $order->load('items');

        // Opening the private link on another device adds it to that browser's list too
        OrderHistory::remember($request, $order);

        return response()
            ->view('orders.show', compact('order'))
            ->withHeaders($this->privateHeaders());
    }

    public function reorder(Order $order)
    {
        $added = 0;

        foreach ($order->items as $item) {
            $product = $item->product_id ? Product::where('is_active', true)->find($item->product_id) : null;

            if ($product) {
                Cart::put($product, Cart::quantityOf($product) + $item->quantity);
                $added++;
            }
        }

        return redirect()->route('cart.index')->with(
            $added ? 'success' : 'error',
            $added
                ? 'Items from order ' . $order->reference . ' were added to your cart. Prices are updated to today\'s prices.'
                : 'Sorry, the products from this order are no longer available.'
        );
    }

    private function privateHeaders(): array
    {
        return [
            'X-Robots-Tag'    => 'noindex, nofollow',
            'Referrer-Policy' => 'no-referrer', // don't leak the token to other sites
            'Cache-Control'   => 'private, no-store',
        ];
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Product;
use App\Support\ProductSearch;
use App\Models\Category;
use App\Models\Company;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class ProductController extends Controller
{
    public function index(Request $request, ?Category $currentCategory = null, ?Company $currentBrand = null)
    {
        // ---- Global price range (for slider/inputs) ----
        $minPriceRaw = Product::where('is_active', true)
            ->min(DB::raw(Product::effectivePriceSql()));

        $maxPriceRaw = Product::where('is_active', true)
            ->max(DB::raw(Product::effectivePriceSql()));

        $minPrice = (int) floor($minPriceRaw ?? 0);
        $maxPrice = (int) ceil($maxPriceRaw ?? 0);

        // ---- Read filters from request ----
        $selectedCategories = (array) $request->input('categories', []);
        $selectedCompanies  = (array) $request->input('companies', []);
        $onSale             = $request->boolean('on_sale', false);

        $currentMin = $request->input('min_price', $minPrice);
        $currentMax = $request->input('max_price', $maxPrice);

        $currentMin = is_numeric($currentMin) ? (float) $currentMin : $minPrice;
        $currentMax = is_numeric($currentMax) ? (float) $currentMax : $maxPrice;

        if ($currentMin > $currentMax) {
            [$currentMin, $currentMax] = [$currentMax, $currentMin];
        }

        $sort    = $request->input('sort', 'default');
        $perPage = (int) $request->input('per_page', 20);
        if (! in_array($perPage, [12, 20, 40, 60])) {
            $perPage = 20;
        }

        // ---- Determine which selected categories actually have those companies ----
        $categoriesWithSelectedCompanies = [];
        if (! empty($selectedCompanies)) {
            $categoriesWithSelectedCompanies = Product::where('is_active', true)
                ->whereIn('company_id', $selectedCompanies)
                ->pluck('category_id')
                ->unique()
                ->all();
        }

        // Intersect with user-selected categories
        $categoriesWithCompanies = [];
        $categoriesWithoutCompanies = [];

        if (! empty($selectedCategories)) {
            $categoriesWithCompanies = array_values(
                array_intersect($selectedCategories, $categoriesWithSelectedCompanies)
            );
            $categoriesWithoutCompanies = array_values(
                array_diff($selectedCategories, $categoriesWithCompanies)
            );
        }

        // ---- Base product query ----
        $query = Product::with(['category', 'company'])
            ->where('is_active', true);

        // Combined category + company logic:
        // - If both categories AND companies are selected:
        //   - For categories that have these companies: filter by company
        //   - For categories WITHOUT companies: include all products from those categories
        if (! empty($selectedCategories) && ! empty($selectedCompanies)) {
            $query->where(function ($q) use (
                $categoriesWithCompanies,
                $categoriesWithoutCompanies,
                $selectedCompanies
            ) {
                // Part 1: categories that have the selected companies
                if (! empty($categoriesWithCompanies)) {
                    $q->where(function ($q2) use ($categoriesWithCompanies, $selectedCompanies) {
                        $q2->whereIn('category_id', $categoriesWithCompanies)
                            ->whereIn('company_id', $selectedCompanies);
                    });
                }

                // Part 2: categories that don't have companies at all
                if (! empty($categoriesWithoutCompanies)) {
                    if (! empty($categoriesWithCompanies)) {
                        $q->orWhereIn('category_id', $categoriesWithoutCompanies);
                    } else {
                        // edge case: only categories without companies
                        $q->whereIn('category_id', $categoriesWithoutCompanies);
                    }
                }
            });
        }
        // Only categories selected (no companies)
        elseif (! empty($selectedCategories)) {
            $query->whereIn('category_id', $selectedCategories);
        }
        // Only companies selected (no categories)
        elseif (! empty($selectedCompanies)) {
            $query->whereIn('company_id', $selectedCompanies);
        }
        // else: no category/company filter

        // ---- Other filters ----
        if ($onSale) {
            $query->where('is_on_sale', true);
        }

        // CAST the bounds: SQLite binds PHP floats as text, which never compares equal/between numbers
        $query->whereRaw(
            '(' . Product::effectivePriceSql() . ') BETWEEN CAST(? AS DECIMAL(10,2)) AND CAST(? AS DECIMAL(10,2))',
            [$currentMin, $currentMax]
        );

        // ---- Sorting ----
        switch ($sort) {
            case 'latest':
                $query->orderBy('created_at', 'desc');
                break;

            case 'price_low_high':
                $query->orderByRaw(Product::effectivePriceSql() . ' asc');
                break;

            case 'price_high_low':
                $query->orderByRaw(Product::effectivePriceSql() . ' desc');
                break;

            default:
                $query->orderBy('name');
                break;
        }

        $products = $query
            ->paginate($perPage)
            ->withQueryString();

        // ---- Sidebar data: categories with counts ----
        $categories = Category::where('is_active', true)
            ->withCount(['products' => function ($q) {
                $q->where('is_active', true);
            }])
            ->orderBy('name')
            ->get();

        // ---- Sidebar data: companies nested under categories ----
        $companyCounts = Product::select(
            'category_id',
            'company_id',
            DB::raw('count(*) as products_count')
        )
            ->whereNotNull('company_id')
            ->where('is_active', true)
            ->groupBy('category_id', 'company_id')
            ->get();

        $companyIds = $companyCounts->pluck('company_id')->unique()->all();

        $companyNames = Company::whereIn('id', $companyIds)
            ->where('is_active', true)
            ->pluck('name', 'id');

        $categoryCompanies = [];

        foreach ($companyCounts as $row) {
            if (! $companyNames->has($row->company_id)) {
                continue;
            }

            $categoryCompanies[$row->category_id][] = [
                'id'    => $row->company_id,
                'name'  => $companyNames[$row->company_id],
                'count' => $row->products_count,
            ];
        }

        return view('shop.index', [
            'currentCategory'    => $currentCategory,
            'currentBrand'       => $currentBrand,
            'products'           => $products,
            'categories'         => $categories,
            'categoryCompanies'  => $categoryCompanies,
            'selectedCategories' => $selectedCategories,
            'selectedCompanies'  => $selectedCompanies,
            'onSale'             => $onSale,
            'minPrice'           => $minPrice,
            'maxPrice'           => $maxPrice,
            'currentMin'         => $currentMin,
            'currentMax'         => $currentMax,
            'sort'               => $sort,
            'perPage'            => $perPage,
        ]);
    }

    public function search(Request $request)
    {
        $queryText   = trim($request->input('q', ''));
        $categoryId  = $request->input('category');

        $products = collect();

        if ($queryText !== '' || $categoryId) {
            $productsQuery = Product::with(['category', 'company'])
                ->where('is_active', true);

            // Forgiving matching: "2.5mm" ↔ "2.5 mm", any word order, brand/category/SKU too (see ProductSearch)
            if ($queryText !== '') {
                ProductSearch::apply($productsQuery, $queryText);
            } else {
                $productsQuery->orderBy('name');

            if (!empty($categoryId)) {
                $productsQuery->where('category_id', $categoryId);
            }

            $products = $productsQuery
                ->paginate(12)

        return view('search.results', [
            'query'    => $queryText,
            'products' => $products,
        ]);
    }

<?php

namespace App\Http\Controllers;

use App\Http\Requests\QuoteRequestForm;
use App\Jobs\SendQuoteWhatsAppAlert;
use App\Mail\NewQuoteRequest;
use App\Models\QuoteRequest;
use App\Models\SiteSetting;
use App\Support\Cart;
use App\Support\PakistaniPhone;
use Illuminate\Support\Facades\Mail;
use Illuminate\Support\Facades\RateLimiter;
use Illuminate\Support\Str;
use Throwable;

// Bulk / trade quote requests (/bulk-quote) – same notifications as orders (email + owner WhatsApp)
class QuoteController extends Controller
{
    public function create()
    {
        $cartItems = Cart::items();

        return view('quotes.create', [
            'cartItems' => $cartItems,
            'cartTotal' => $cartItems->sum('cart_subtotal'),
        ]);
    }

    public function store(QuoteRequestForm $request)
    {
        // Honeypot: pretend it worked, store nothing
        if ($request->filled('website')) {
            return redirect()->route('quotes.thanks');
        }

        $phone    = PakistaniPhone::normalize($request->validated('phone'));
        $ipKey    = 'quote:ip:' . $request->ip();
        $phoneKey = 'quote:phone:' . $phone;

        if (RateLimiter::tooManyAttempts($ipKey, 5) || RateLimiter::tooManyAttempts($phoneKey, 3)) {
            return back()
                ->withErrors(['form' => 'You have sent several requests recently. Please wait a while, or call/WhatsApp us directly.'])
                ->withInput();
        }

        $attachmentPath = $attachmentName = null;
        if ($file = $request->file('attachment')) {
            // Random name on the private disk – never served directly
            $attachmentPath = $file->storeAs('quotes', Str::random(40) . '.' . $file->guessExtension(), QuoteRequest::ATTACHMENT_DISK);
            $attachmentName = Str::limit(preg_replace('/[^\w.\- ]+/', '_', $file->getClientOriginalName()), 120, '');
        }

        $cartItems = $request->boolean('include_cart')
            ? Cart::items()->map(fn ($p) => [
                'product_id'   => $p->id,
                'product_name' => $p->name,
                'sku'          => $p->sku,
                'quantity'     => $p->cart_quantity,
                'unit_price'   => $p->cart_price,
                'line_total'   => $p->cart_subtotal,
            ])->values()->all()
            : null;

        $quote = QuoteRequest::create([
            'name'            => $request->validated('name'),
            'phone'           => $phone,
            'email'           => $request->validated('email'),
            'business_name'   => $request->validated('business_name'),
            'city'            => $request->validated('city'),
            'items'           => (string) $request->validated('items'),
            'cart_items'      => $cartItems ?: null,
            'attachment_path' => $attachmentPath,
            'attachment_name' => $attachmentName,
            'ip_address'      => $request->ip(),
            'user_agent'      => Str::limit((string) $request->userAgent(), 1000, ''),
        ]);

        RateLimiter::hit($ipKey, 3600);
        RateLimiter::hit($phoneKey, 3600);

        // A notification problem must never fail the customer's request (failures land in the Error log)
        $notify = function (callable $send) {
            try {
                $send();
            } catch (Throwable $e) {
                report($e);
            }
        };

        $adminEmail = config('orders.admin_email') ?: SiteSetting::current()->contact_recipient_email;
        if ($adminEmail) {
            $notify(fn () => Mail::to($adminEmail)->queue(new NewQuoteRequest($quote)));
        }
        $notify(fn () => SendQuoteWhatsAppAlert::dispatch($quote));

        return redirect()->route('quotes.thanks')->with('quote_reference', $quote->reference);
    }

    public function thanks()
    {
        return response()
            ->view('quotes.thanks', ['reference' => session('quote_reference')])
            ->header('X-Robots-Tag', 'noindex, nofollow');
    }
}

    // Private/utility pages stay out of search results; non-production sites are hidden completely
    public function robots()
    {
        $lines = ['User-agent: *'];

        if (app()->isProduction()) {
            foreach (['/admin', '/cart', '/checkout', '/my-orders', '/orders/', '/wishlist', '/search'] as $path) {
                $lines[] = 'Disallow: ' . $path;
            }
            $lines[] = '';
            $lines[] = 'Sitemap: ' . route('sitemap');
        } else {
            $lines[] = 'Disallow: /';
        }

        return response(implode("\n", $lines) . "\n", 200, ['Content-Type' => 'text/plain; charset=UTF-8']);
    }

    // All public pages Google should know about (cached for an hour)
    public function sitemap()
    {
        $xml = Cache::remember('sitemap.xml', 3600, function () {
            $urls = collect([
                ['loc' => route('home'), 'priority' => '1.0'],
                ['loc' => route('shop.index'), 'priority' => '0.9'],
                ['loc' => route('contact.show'), 'priority' => '0.5'],
            ]);

            Category::where('is_active', true)->orderBy('sort_order')->get(['slug', 'updated_at'])
                ->each(fn ($c) => $urls->push(['loc' => route('category.show', $c->slug), 'lastmod' => $c->updated_at, 'priority' => '0.8']));

            Company::where('is_active', true)->whereHas('products', fn ($q) => $q->where('is_active', true))->get(['slug', 'updated_at'])
                ->each(fn ($b) => $urls->push(['loc' => route('brand.show', $b->slug), 'lastmod' => $b->updated_at, 'priority' => '0.6']));

            Product::where('is_active', true)->latest('updated_at')->get(['slug', 'updated_at'])
                ->each(fn ($p) => $urls->push(['loc' => route('product.show', $p->slug), 'lastmod' => $p->updated_at, 'priority' => '0.7']));

            return view('seo.sitemap', ['urls' => $urls])->render();
        });

        return response($xml, 200, ['Content-Type' => 'application/xml; charset=UTF-8']);

<?php

namespace App\Http\Controllers;

use App\Models\Product;
use Illuminate\Http\Request;

class WishlistController extends Controller
{
    public function index()
    {
        $wishlist = session('wishlist', []);

        if (empty($wishlist)) {
            $products = collect();
        } else {
            $products = Product::with(['category', 'company'])
                ->whereIn('id', $wishlist)
                ->where('is_active', true)
                ->get();
        }

        return view('wishlist.index', compact('products'));
    }

    public function add(Request $request, Product $product)
    {
        $wishlist = session('wishlist', []);

        if (in_array($product->id, $wishlist)) {
            // toggle off
            $wishlist = array_values(array_diff($wishlist, [$product->id]));
            $inWishlist = false;
            $msg = 'Removed from wishlist';
        } else {
            $wishlist[] = $product->id;
            $inWishlist = true;
            $msg = 'Added to wishlist';
        }

        session(['wishlist' => $wishlist]);

        $wishlistCount = count($wishlist);

        if ($request->expectsJson()) {
            return response()->json([
                'success'         => true,
                'message'         => $msg,
                'wishlist_count'  => $wishlistCount,
                'in_wishlist'     => $inWishlist,
                'product_id'      => $product->id,
            ]);
        }

        return back()->with('success', $msg);
    }

    public function remove(Product $product)
    {
        $wishlist = session('wishlist', []);
        $wishlist = array_values(array_filter($wishlist, fn ($id) => (int)$id !== (int)$product->id));

        session(['wishlist' => $wishlist]);

        return back()->with('success', 'Product removed from wishlist.');
    }

    public function clear()
    {
        session()->forget('wishlist');

        return back()->with('success', 'Wishlist cleared.');
    }
}

    $ga4   = content('seo.ga4_id');

@if($ga4)

    <script>

        if(!f._fbq)f._fbq=n;n.push=n;n.loaded=!0;n.version='2.0';n.queue=[];t=b.createElement(e);t.async=!0;t.src=v;
        s=b.getElementsByTagName(e)[0];s.parentNode.insertBefore(t,s)}(window,document,'script','https://connect.facebook.net/en_US/fbevents.js');
        fbq('init', @js($pixel));

    </script>
@endif
<script>
    // event: 'add_to_cart' | 'begin_checkout' | 'lead'   data: { value, items: [{ id, name, price, quantity }], kind }
    window.ueTrack = function (event, data) {
        data = data || {};
        var items = data.items || [];
        var value = Number(data.value || 0);
        try {
            if (typeof window.gtag === 'function') {
                var gaName = { add_to_cart: 'add_to_cart', begin_checkout: 'begin_checkout', lead: 'generate_lead' }[event];
                if (gaName) {
                    window.gtag('event', gaName, {
                        currency: 'PKR', value: value, lead_type: data.kind,
                        items: items.map(function (i) { return { item_id: String(i.id), item_name: i.name, price: Number(i.price), quantity: Number(i.quantity || 1) }; })
                    });
                }
            }
            if (typeof window.fbq === 'function') {
                var fbName = { add_to_cart: 'AddToCart', begin_checkout: 'InitiateCheckout', lead: 'Lead' }[event];
                if (fbName) {
                    window.fbq('track', fbName, {
                        currency: 'PKR', value: value, content_type: 'product',
                        content_ids: items.map(function (i) { return String(i.id); }),
                        content_name: data.kind || (items[0] && items[0].name) || undefined
                    });
                }
            }
        } catch (e) { /* tracking must never break the page */ }
    };
</script>

// One entry in the admin activity log (see App\Models\Concerns\LogsAdminActivity)
class AdminActivity extends Model
{
    use Prunable;

    public const UPDATED_AT = null;

    protected $fillable = [
        'user_id',
        'user_name',
        'action',
        'subject_type',
        'subject_id',
        'subject_label',
        'changes',
        'ip_address',
    ];

    protected function casts(): array
    {
        return ['changes' => 'array'];
    }

    public function user()
    {
        return $this->belongsTo(User::class);
    }

    // Removed after a year by the daily `model:prune`
    public function prunable(): Builder
    {
        return static::where('created_at', '<', now()->subYear());
    }

    // Record an action by the logged-in admin (no-op for customers, console and seeders)
    public static function record(string $action, ?string $label = null, ?Model $subject = null, ?array $changes = null): void
    {
        $user = auth()->user();

        if (! ($user instanceof User) || (app()->runningInConsole() && ! app()->runningUnitTests())) {
            return;
        }

        static::create([
            'user_id'       => $user->id,
            'user_name'     => $user->name,
            'action'        => $action,
            'subject_type'  => $subject ? class_basename($subject) : null,
            'subject_id'    => $subject?->getKey(),
            'subject_label' => $label ? mb_substr($label, 0, 255) : null,
            'changes'       => $changes ?: null,
            'ip_address'    => request()->ip(),
        ]);
    }
}

<?php

namespace App\Models\Concerns;

use App\Support\ImageVariants;
use Illuminate\Support\Facades\Storage;

/**
 * For models with uploaded images: creates the WebP copies after saving, and removes files
 * (original + copies) that were replaced or whose record was deleted, so storage doesn't fill up.
 * The using model lists its image columns in imageAttributes() (string or array-of-paths columns).
 */
trait HasImageVariants
{
    /** @return list<string> */
    abstract protected function imageAttributes(): array;

    public static function bootHasImageVariants(): void
    {
        static::saved(function ($model) {
            $current  = $model->imagePaths();
            $previous = $model->imagePaths(original: true);

            foreach ($current as $path) {
                ImageVariants::generate($path);
            }

            // On create there are no previous paths, so nothing is removed
            foreach (array_diff($previous, $current) as $removed) {
                self::removeImageFile($removed);
            }
        });

        static::deleted(function ($model) {
            foreach ($model->imagePaths() as $path) {
                self::removeImageFile($path);
            }
        });
    }

    /** @return list<string> */
    public function imagePaths(bool $original = false): array
    {
        $paths = [];

        foreach ($this->imageAttributes() as $attribute) {
            $value = $original ? $this->getOriginal($attribute) : $this->getAttribute($attribute);

            if (is_string($value) && str_starts_with($value, '[')) {
                $value = json_decode($value, true);
            }

            foreach ((array) $value as $path) {
                // Only files uploaded through the admin (public disk) – never legacy "storage/…" or external URLs
                if (is_string($path) && $path !== '' && ! str_starts_with($path, 'storage/') && ! preg_match('/^https?:/i', $path)) {
                    $paths[] = $path;
                }
            }
        }

        return array_values(array_unique($paths));
    }

    private static function removeImageFile(string $path): void
    {
        Storage::disk('public')->delete($path);
        ImageVariants::delete($path);
    }
}


--- FILE: app/Models/Concerns/LogsAdminActivity.php ---

<?php

namespace App\Models\Concerns;

use App\Models\AdminActivity;
use BackedEnum;

/**
 * Logs create/update/delete of this model by a logged-in admin into the activity log.
 * Secrets and noise (timestamps, tokens, passwords) are never stored; long texts are shortened.
 */
trait LogsAdminActivity
{
    private static array $neverLogged = [
        'created_at', 'updated_at', 'password', 'remember_token', 'public_token',
        'app_authentication_secret', 'app_authentication_recovery_codes', 'last_login_at',
    ];

    public static function bootLogsAdminActivity(): void
    {
        static::created(fn ($model) => AdminActivity::record('created', $model->activityLabel(), $model));

        static::updated(function ($model) {
            $changes = [];

            foreach (array_keys($model->getChanges()) as $field) {
                if (in_array($field, self::$neverLogged, true)) {
                    continue;
                }

                $changes[$field] = [
                    'old' => self::forLog($model->getOriginal($field)),
                    'new' => self::forLog($model->getAttribute($field)),
                ];
            }

            if ($changes) {
                AdminActivity::record('updated', $model->activityLabel(), $model, $changes);
            }
        });

        static::deleted(fn ($model) => AdminActivity::record('deleted', $model->activityLabel(), $model));
    }

    // e.g. "Product: LED Bulb 12W", "Order: UE-100004"
    public function activityLabel(): string
    {
        $name = $this->reference ?? $this->name ?? $this->title ?? $this->email ?? ('#' . $this->getKey());

        return class_basename($this) . ': ' . $name;
    }

    private static function forLog(mixed $value): mixed
    {
        if ($value instanceof BackedEnum) {
            return $value->value;
        }

        if ($value instanceof \DateTimeInterface) {
            return $value->format('Y-m-d H:i');
        }

        if (is_array($value)) {
            return mb_strimwidth(json_encode($value, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES), 0, 200, '…');
        }

        if (is_string($value)) {
            return mb_strimwidth(strip_tags($value), 0, 200, '…');
        }

        return $value;
    }
}

    {
        return [
            'status'                 => OrderStatus::class,
            'subtotal'               => 'decimal:2',
            'contacted_at'           => 'datetime',
            'confirmed_at'           => 'datetime',
            'dispatched_at'          => 'datetime',
            'delivered_at'           => 'datetime',
            'cancelled_at'           => 'datetime',
            'owner_whatsapp_sent_at' => 'datetime',
        ];
    }

    protected static function booted(): void
    {
        static::creating(function (Order $order) {
            $order->public_token ??= Str::random(48);
            $order->status ??= OrderStatus::New;
        });

        // Reference is derived from the id, so it is unique and short (UE-100042)

            $order->saveQuietly();

    {

        // Keep a simple running log in the admin note
        $line = now()->format('d M Y H:i') . ' – ' . $status->getLabel() . ($by ? " by {$by}" : '') . ($note ? ": {$note}" : '');
        $this->admin_note = trim(($this->admin_note ? $this->admin_note . "\n" : '') . $line);

        if ($column = $status->timestampColumn()) {
            $this->{$column} ??= now();
        }

        $this->save();
    }

    // When the order reached a given step (null = not reached yet)
    public function reachedAt(OrderStatus $status)
    {
        $column = $status->timestampColumn();

        return $column ? $this->{$column} : $this->created_at;
    }

    {
        return PakistaniPhone::display($this->customer_phone);
    }

    // 0300 ***4567 – for customer-facing pages reachable by link
    public function maskedPhone(): string
    {
        $display = $this->displayPhone();

        return preg_match('/^(\d{4}) \d{3}(\d{4})$/', $display, $m) ? "{$m[1]} ***{$m[2]}" : $display;
    }

    {
        $url = 'https://wa.me/' . ltrim($this->customer_phone, '+');

        return $text ? $url . '?text=' . rawurlencode($text) : $url;
    }

        'size_label',
        'customer_note',
        'price',
        'sale_price',
        'is_on_sale',
        'stock',
        'is_active',
        'is_featured',
        'short_description',
        'description',
        'image',
        'gallery',
        'meta_title',
        'meta_description',

    protected function casts(): array
    {
        return [
            'gallery'     => 'array',
            'is_on_sale'  => 'boolean',
            'is_active'   => 'boolean',
            'is_featured' => 'boolean',
        ];
    }

    public function category()
    {
        return $this->belongsTo(Category::class);
    }

    public function company()
    {
        return $this->belongsTo(Company::class);
    }

    // Single price rule: sale price only counts when the product is flagged on sale.
    // Keep effectivePriceSql() in sync with this for DB filtering/sorting.
    public function getDisplayPrice(): float
    {
        return $this->is_on_sale && $this->sale_price
            ? (float) $this->sale_price
            : (float) $this->price;
    }

    // Other visible products in the same size group (e.g. the 180 m coil of the same wire), cheapest first
    public function otherSizes()
    {
        if (blank($this->size_group)) {
            return collect();
        }

        return static::where('size_group', $this->size_group)
            ->where('is_active', true)
            ->whereKeyNot($this->getKey())
            ->orderByRaw(static::effectivePriceSql())
            ->get();
    }

    // Short text for size buttons
    public function sizeLabel(): string
    {
        return $this->size_label ?: $this->name;
    }

    // Main image (WebP copy of the requested size when available) or null
    public function imageUrl(string $size = 'thumb'): ?string
    {
        return ImageVariants::url($this->image, $size);
    }

    // Main image followed by gallery images
    public function imageUrls(string $size = 'large'): array
    {
        return collect([$this->image])
            ->merge($this->gallery ?? [])
            ->filter()
            ->unique()
            ->map(fn (string $path) => ImageVariants::url($path, $size))
            ->values()
            ->all();
    }

    // SQL version of getDisplayPrice() for whereBetween / orderBy / min / max
    public static function effectivePriceSql(): string
    {
        return 'CASE WHEN is_on_sale = 1 AND sale_price > 0 THEN sale_price ELSE price END';
    }
}

    {
        if (blank($value)) {

        }

        if (preg_match('/src=["\']([^"\']+)["\']/i', $value, $m)) {
            $value = html_entity_decode($m[1]);
        }

        $value = trim($value);

        return str_starts_with($value, 'https://www.google.com/maps/embed') ? $value : null;
    }

    public static function storeName(): string
    {
        try {
